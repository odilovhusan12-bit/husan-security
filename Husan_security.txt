@app.route("/", methods=["GET", "POST"])
def home():
    result = ""
    kind = ""

    if request.method == "POST":
        action = request.form.get("action", "")
        kind = action

        if action == "url":
            value = request.form.get("url", "").strip()
            parsed = urlparse(value)
            warnings = []

            if parsed.scheme not in ("http", "https") or not parsed.hostname:
                result = "URL formati noto‘g‘ri."
            else:
                if parsed.scheme != "https":
                    warnings.append("HTTPS ishlatilmayapti.")
                if "@" in parsed.netloc:
                    warnings.append("Manzilda @ belgisi mavjud.")
                if len(value) > 100:
                    warnings.append("URL juda uzun.")
                if parsed.hostname.startswith("xn--") or ".xn--" in parsed.hostname:
                    warnings.append("Domen Punycode formatida.")

                result = (
                    "E'tibor talab qiladigan belgilar: " + " ".join(warnings)
                    if warnings else
                    "Oddiy tekshiruvda shubhali belgi topilmadi. Bu xavfsizlik kafolati emas."
                )

        elif action == "domain":
            domain = request.form.get("domain", "").strip()
            domain = domain.removeprefix("https://").removeprefix("http://")
            domain = domain.split("/")[0].split(":")[0]

            try:
                if not domain or " " in domain:
                    raise ValueError()
                addresses = sorted({
                    item[4][0] for item in socket.getaddrinfo(
                        domain, None, type=socket.SOCK_STREAM
                    )
                })
                result = "Domen: " + domain + "\nIP manzillar: " + ", ".join(addresses)
            except (socket.gaierror, ValueError, OSError):
                result = "Domen aniqlanmadi yoki DNS orqali tekshirib bo‘lmadi."

        elif action == "file":
            uploaded = request.files.get("file")
            if not uploaded or not uploaded.filename:
                result = "Fayl tanlanmadi."
            else:
                data = uploaded.stream.read(20 * 1024 * 1024 + 1)
                if len(data) > 20 * 1024 * 1024:
                    result = "Fayl hajmi 20 MB dan oshmasligi kerak."
                else:
                    digest = hashlib.sha256(data).hexdigest()
                    extension = os.path.splitext(uploaded.filename)[1].lower()
                    warnings = []
                    if extension in (".exe", ".dll", ".bat", ".cmd", ".scr", ".ps1"):
                        warnings.append("Fayl bajariladigan yoki skript turiga kiradi.")
                    if data.startswith(b"MZ"):
                        warnings.append("Windows PE fayliga xos MZ belgisi topildi.")

                    result = (
                        f"Fayl nomi: {uploaded.filename}\n"
                        f"Hajmi: {len(data)} bayt\n"
                        f"SHA-256: {digest}\n"
                        f"Tekshiruv: {'; '.join(warnings) if warnings else 'Faqat asosiy belgilar tekshirildi.'}"
                    )

    return render_template_string(HTML, result=result, kind=kind)
